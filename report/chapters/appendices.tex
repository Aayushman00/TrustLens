\chapter{Reproducing the Experiments}\label{app:repro}
Commands are run from \code{backend/\allowbreak } with the project virtual environment; training uses \code{USE\_TF=0}. The Docker stack must be running for the API-based runs (rebuild \code{api} and \code{worker} after backend changes). Seeds and output folders are those named in the corresponding plan.
\begin{lstlisting}
# controlled suite (data, training, evaluation through the API)
python -m app.scripts.prepare_flawed_suite_data
python -m app.scripts.train_flawed_suite --all
python -m app.scripts.run_flawed_suite_eval
python -m app.scripts.repeat_eval            # repeated runs
python -m app.scripts.compare_ground_truth   # apply the frozen rules

# L3 replay
python -m app.scripts.replay_l3_disclosure_gaps <new_out_dir>

# L5 confirmatory fairness (fresh seed, fresh data)
python -m app.scripts.prepare_flawed_suite_data --fairness-confirm \
    --seed 20261003 --out ../results/fairness_confirm_20261003
python -m app.scripts.train_flawed_suite --fairness-confirm \
    ../results/fairness_confirm_20261003 --seed 20261003
python -m app.scripts.run_fairness_confirm

# L6 severity sweeps
python -m app.scripts.run_defect_severity --out ../results/defect_severity_20261003

# L10 human analysis (stdlib only, from the repository root)
python results/l10_human_form/analyze_human_responses.py

# external Hub audit
python -m app.scripts.run_external_hub_validation select
python -m app.scripts.run_external_hub_validation run
python -m app.scripts.run_external_hub_validation profile

# figures and this report
cd report
python make_figures.py          # pilot figures
python make_figures_final.py    # L1, L5, L6, external-audit figures
tectonic main.tex               # or: pdflatex main.tex  (twice)
\end{lstlisting}

\chapter{Identifiers}
\begin{table}[H]
\centering\footnotesize
\caption{Risk, disclosure-gap and gate identifiers used in the report.}
\begin{tabularx}{\textwidth}{@{}p{4.3cm}L@{}}
\toprule
\textbf{Identifier} & \textbf{Meaning}\\
\midrule
\code{F-FAIR-EOPP} & Equal-opportunity (TPR) gap above $\varepsilon$ with its CI lower bound above $\varepsilon$ (risk).\\
\code{F-FAIR-PERF} & Multiclass worst-group accuracy gap (risk; not exercised here).\\
\code{R-ROB-PERT} & Accuracy drop under perturbation above $\varepsilon_{drop}$ with CI (risk).\\
\code{I-INT-BYTES-DIVERGE} & Artifact bytes differ from the manifest (risk).\\
\code{I-INT-LISTING-DRIFT} & File listing drift (risk).\\
\code{E-DOC-CONTRADICTION} & Card contradicts itself or its licence (risk).\\
\code{S-GOV-DISCLOSURE-GAP} & A safety disclosure item is missing (gap).\\
\code{E-DOC-INCOMPLETE} & A required documentation section is missing (gap).\\
\code{I-INT-REV-UNPINNED}, \code{I-INT-MANIFEST-MISSING}, \code{I-INT-LICENSE-UNDISCLOSED} & Revision not pinned, no manifest or file listing, no licence in metadata (gaps).\\
\code{G-FAIR-CI-WIDE}, \code{G-FAIR-N-GROUP}, \code{G-FAIR-POSITIVES}, \code{G-FAIR-CI-MISSING} & Fairness gates (insufficient evidence).\\
\code{G-ROB-PERT-COVERAGE} & Robustness gate: perturbation coverage must be 1.0.\\
\code{G-INT-ARTIFACT-INCOMPLETE}, \code{G-INT-LISTING-UNVERIFIED} & Integrity gates (not a risk).\\
\bottomrule
\end{tabularx}
\end{table}
